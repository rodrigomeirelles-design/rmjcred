function verifyAdminLogin(email, password) {

  if (email !== 'rodrigomeirelles@gmail.com') return { success: false, message: 'Usuário não autorizado.' };

  const props = PropertiesService.getScriptProperties();

  const storedHash = props.getProperty('ADMIN_PASS_HASH');

  if (storedHash !== computeSHA256_(password)) return { success: false, message: 'Senha incorreta.' };

  const isFirstAccess = props.getProperty('ADMIN_FIRST_ACCESS') === 'true';

  return { success: true, firstAccess: isFirstAccess, token: generateSessionToken_(email) };

}



function changeAdminPassword(email, currentPassword, newPassword) {

function generateSessionToken_(email) {

  const token = Utilities.getUuid() + '-' + new Date().getTime();

  CacheService.getScriptCache().put('session_' + token, email, 1800);

  return token;

}



function validateSession_(token) {

  if (!token) return false;

  return CacheService.getScriptCache().get('session_' + token) === 'rodrigomeirelles@gmail.com';

}



function computeSHA256_(input) {

  const rawHash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input, Utilities.Charset.UTF_8);

  let output = '';

  for (let i = 0; i < rawHash.length; i++) {

    let value = rawHash[i];

    if (value < 0) value += 256;

    let byteString = value.toString(16);

    if (byteString.length == 1) byteString = '0' + byteString;

    output += byteString;

  }

  return output;

}

function fetchAllPropostasAdmin(sessionToken) {

  if (!validateSession_(sessionToken)) throw new Error('Sessão expirada. Inicie sessão novamente.');

  const headers = getHeaders_(SHEETS.RESP);

  const respSh = ss_().getSheetByName(SHEETS.RESP);

  if (!headers.length || respSh.getLastRow() < 2) return [];

  

  const data = respSh.getRange(2, 1, respSh.getLastRow() - 1, headers.length).getValues();

  const iId = headers.indexOf('ID'), iCnpj = headers.indexOf('emp_cnpj'), iRazao = headers.indexOf('emp_razao'), iWhen = headers.indexOf('Data/Hora');

  const isUUID = v => typeof v === 'string' && /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(v);

  

  return data.filter(r => isUUID(r[iId])).map(r => ({

    id: String(r[iId]),

    cnpj: String(r[iCnpj]),

    razao: String(r[iRazao]),

    when: r[iWhen] instanceof Date ? r[iWhen].getTime() : new Date(r[iWhen]).getTime()

  })).sort((a, b) => b.when - a.when);

}



function fetchFullPropostaDataAdmin(sessionToken, propostaId) {

  if (!validateSession_(sessionToken)) throw new Error('Acesso negado.');

  const headers = getHeaders_(SHEETS.RESP);

  const respSh = ss_().getSheetByName(SHEETS.RESP);

  const data = respSh.getRange(2, 1, respSh.getLastRow() - 1, headers.length).getValues();

  const row = data.find(r => r[headers.indexOf('ID')] === propostaId);

  if (!row) throw new Error('Proposta não encontrada.');

  

  const obj = {};

  headers.forEach((k, i) => {

    let val = row[i];

    if (val instanceof Date) {

      obj[k] = Utilities.formatDate(val, 'America/Sao_Paulo', 'dd/MM/yyyy HH:mm');

    } else {

      obj[k] = String(val == null ? '' : val).replace(/</g, '&lt;').replace(/>/g, '&gt;');

    }

  });

  return obj;

}

function submitProposta(payload) {

  const lock = LockService.getScriptLock();

  try { lock.waitLock(30000); } catch(e) { return {ok:false, errors:['Servidor ocupado.']}; }

  try {

    const clean  = sanitizePayload_(payload);

    const errors = validatePayload_(clean);

    if (errors.length) return {ok:false, errors:errors};

    

    const id = Utilities.getUuid(), now = new Date();

    clean._meta = {id, when:now, version:APP_VERSION};

    

    gravarRespostas_(clean);

    gravarLegiveis_(clean);

    gravarPerguntasRespostas_(clean);

    SpreadsheetApp.flush();

    

    dispararNotificacaoEmail_(clean);

    return {ok:true, id};

  } catch(e) {

    return {ok:false, errors:['Erro interno: '+e.message]};

  } finally { lock.releaseLock(); }

}

/* ── SANITIZAÇÃO E VALIDAÇÃO ──────────────────────────────────────── */

function sanitizeString_(v) {

  if (v==null) return '';

  let s=String(v).replace(/[\u0000-\u001F\u007F]/g,'').replace(/<[^>]*>/g,'').trim();

  if (/^[=+\-@|`]/.test(s)) s="'"+s;

  return s.length>500?s.substring(0,500):s;

}



function sanitizePayload_(p) {

  const out={empresa:{},bancario:{},observacoes:{},socios:[]};

  getSchema_().forEach(f=>{

    const sec=f.sec.toLowerCase(); out[sec]=out[sec]||{};

    out[sec][f.key]=sanitizeString_(((p[f.sec]||p[sec])||{})[f.key]);

  });

  (Array.isArray(p.socios)?p.socios:[]).forEach(s=>{

    const so={dados:{},conjuge:null};

    getSocioSchema_().forEach(f=>so.dados[f.key]=sanitizeString_((s.dados||{})[f.key]));

    if (s.conjuge&&ESTADOS_CIVIS_COM_CONJUGE.indexOf(so.dados.estado_civil)>=0) {

      so.conjuge={};

      getConjugeSchema_().forEach(f=>so.conjuge[f.key]=sanitizeString_(s.conjuge[f.key]));

    }

    out.socios.push(so);

  });

  return out;

}



function validatePayload_(p) {

  const errs=[];

  const chk=(obj,f,pre)=>{

    const v=(obj||{})[f.key]; const empty=v===''||v==null;

    if(f.required&&empty){errs.push(pre+f.label+' é obrigatório.');return;}

    if(empty)return;

    switch(f.type){

      case 'cpf':    if(!validaCPF_(v))  errs.push(pre+f.label+' inválido.');break;

      case 'cnpj':   if(!validaCNPJ_(v)) errs.push(pre+f.label+' inválido.');break;

      case 'email':  if(!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v)) errs.push(pre+f.label+' inválido.');break;

      case 'phone':  if(onlyDigits_(v).length<10) errs.push(pre+f.label+' inválido.');break;

      case 'cep':    if(onlyDigits_(v).length!==8) errs.push(pre+f.label+' inválido.');break;

      case 'uf':     if(!/^[A-Z]{2}$/.test(v)) errs.push(pre+f.label+' inválido.');break;

      case 'int':    if(!/^\d+$/.test(v)) errs.push(pre+f.label+' deve ser inteiro.');break;

      case 'money':  if(parseMoney_(v)<0) errs.push(pre+f.label+' inválido.');break;

      case 'percent':{const n=parseFloat(String(v).replace(',','.'));if(isNaN(n)||n<0||n>100)errs.push(pre+f.label+' deve ser 0–100.');}break;

    }

  };

  getSchema_().forEach(f=>chk(p[f.sec.toLowerCase()],f,''));

  if(!p.socios||!p.socios.length){errs.push('Cadastre ao menos 1 sócio.');return errs;}

  return errs;

}



function validaCPF_(c){const s=onlyDigits_(c);if(s.length!==11||/^(\d)\1+$/.test(s))return false;let x=0,r;for(let i=1;i<=9;i++)x+=+s[i-1]*(11-i);r=(x*10)%11;if(r>=10)r=0;if(r!==+s[9])return false;x=0;for(let i=1;i<=10;i++)x+=+s[i-1]*(12-i);r=(x*10)%11;if(r>=10)r=0;return r===+s[10];}

function validaCNPJ_(c){const s=onlyDigits_(c);if(s.length!==14||/^(\d)\1+$/.test(s))return false;const calc=b=>{const w=b.length===12?[5,4,3,2,9,8,7,6,5,4,3,2]:[6,5,4,3,2,9,8,7,6,5,4,3,2];let x=0;for(let i=0;i<b.length;i++)x+=+b[i]*w[i];const r=x%11;return r<2?0:11-r;};const d1=calc(s.slice(0,12)),d2=calc(s.slice(0,12)+d1);return d1===+s[12]&&d2===+s[13];}

function onlyDigits_(s){return String(s||'').replace(/\D/g,'');}

function parseMoney_(s){if(typeof s==='number')return s;const t=String(s||'').replace(/[^\d,.-]/g,'').replace(/\./g,'').replace(',','.');const n=parseFloat(t);return isNaN(n)?0:n;}

function ss_(){return SHEET_ID?SpreadsheetApp.openById(SHEET_ID):SpreadsheetApp.getActiveSpreadsheet();}



function getHeaders_(shName){

  const cache=CacheService.getScriptCache(),ck='hdr_'+shName;

  const c=cache.get(ck);if(c)return JSON.parse(c);

  const sh=ss_().getSheetByName(shName);

  if(!sh||sh.getLastColumn()===0)return [];

  const h=sh.getRange(1,1,1,sh.getLastColumn()).getValues()[0];

  cache.put(ck,JSON.stringify(h),21600);return h;

}

function invalidateHeaderCache_(){const c=CacheService.getScriptCache();Object.values(SHEETS).forEach(n=>c.remove('hdr_'+n));}



function ensureSheets_(){

  const s=ss_(); Object.values(SHEETS).forEach(n=>{if(!s.getSheetByName(n))s.insertSheet(n);});

}



function buildHeaderRow_(){

  const ss=ss_(),base=['Data/Hora','ID','Versão'],emp=getSchema_();

  const hT=base.concat(emp.map(f=>f.key)),hL=base.concat(emp.map(f=>f.label));

  for(let i=1;i<=5;i++){

    getSocioSchema_().forEach(f=>{hT.push(`socio${i}_${f.key}`);hL.push(`Sócio ${i} – ${f.label}`);});

    getConjugeSchema_().forEach(f=>{hT.push(`socio${i}_conjuge_${f.key}`);hL.push(`Cônjuge ${i} – ${f.label}`);});

  }

  const tec=ss.getSheetByName(SHEETS.RESP),leg=ss.getSheetByName(SHEETS.LEG),pr=ss.getSheetByName(SHEETS.PR);

  if(tec.getLastRow()===0)tec.appendRow(hT);

  if(leg.getLastRow()===0)leg.appendRow(hL);

  if(pr.getLastRow()===0) pr.appendRow(['Data/Hora','ID','Seção','Campo','Pergunta','Resposta']);

  invalidateHeaderCache_();

}

function buildCell_(k,p){

  if(k==='Data/Hora')return p._meta.when;if(k==='ID')return p._meta.id;if(k==='Versão')return p._meta.version;

  const f=getSchema_().find(x=>x.key===k);if(f)return(p[f.sec.toLowerCase()]||{})[k]||'';

}

}
